## `entry.go` — CacheEntry

```go
package cache

import (
    "encoding/json"
    "net/http"
    "time"
)

// Entry holds a cached API response.
type Entry struct {
    StatusCode  int         `json:"status_code"`
    Headers     http.Header `json:"headers"`
    Body        []byte      `json:"body"`
    CachedAt    time.Time   `json:"cached_at"`
    RequestHash string      `json:"request_hash"` // SHA256 hex al request body
}

// serialize converts Entry to bytes for storage.
func (e *Entry) serialize() ([]byte, error) {
    return json.Marshal(e)
}

// deserialize restores Entry from bytes.
func deserialize(data []byte) (*Entry, error) {
    var e Entry
    if err := json.Unmarshal(data, &e); err != nil {
        return nil, err
    }
    return &e, nil
}
```

var sensitiveHeaders = map[string]bool{

func sanitizeHeaders(h http.Header) http.Header {

    for k := range sensitiveHeaders {
        out.Del(k)
    }
    return out

## `cache.go` — Cache Manager (interfata publica)

Aceasta este singura structura expusa catre `proxy/handler.go`.

```go
package cache

import (
    "context"
    "log/slog"
)

// Config holds cache configuration.
type Config struct {
    DB             interface{} // *sql.DB — primit din stats.Store
    TTL            time.Duration // default: 24h
    SemanticThreshold float64  // default: 0.92 cosine similarity
    EmbeddingProvider EmbeddingProvider
    MaxEntries     int         // default: 10_000 (LRU eviction dupa asta)
}

// Cache orchestrates L1 and L2 lookup/store.
type Cache struct {
    config   Config
    exact    *ExactCache
    semantic *SemanticCache
}

// New creates a Cache with both layers initialized.
func New(config Config) (*Cache, error) {
    if config.TTL == 0 {
        config.TTL = 24 * time.Hour
    }
    if config.SemanticThreshold == 0 {
        config.SemanticThreshold = 0.92
    }
    if config.MaxEntries == 0 {
        config.MaxEntries = 10_000
    }

    exact, err := newExactCache(config)
    if err != nil {
        return nil, fmt.Errorf("cache: init L1: %w", err)
    }

    semantic, err := newSemanticCache(config)
    if err != nil {
        return nil, fmt.Errorf("cache: init L2: %w", err)
    }

    return &Cache{config: config, exact: exact, semantic: semantic}, nil
}

// Result holds a cache lookup result.
type Result struct {
    Entry      *Entry
    Hit        bool
    Level      string  // "L1" | "L2"
    Similarity float64 // 1.0 pentru L1, cosine score pentru L2
}

// Get performs L1 then L2 lookup.
// Returns Result with Hit=false if no cache entry found.
func (c *Cache) Get(ctx context.Context, requestBody []byte) Result {
    // L1: exact match
    if entry, ok := c.exact.Get(ctx, requestBody); ok {
        slog.Debug("cache L1 hit")
        return Result{Entry: entry, Hit: true, Level: "L1", Similarity: 1.0}
    }

    // L2: semantic similarity
    if entry, score, ok := c.semantic.Get(ctx, requestBody); ok {
        slog.Debug("cache L2 hit", "similarity", score)
        return Result{Entry: entry, Hit: true, Level: "L2", Similarity: score}
    }

    return Result{Hit: false}
}

// Set stores entry in both L1 and L2.
// L2 storage este async (nu blocheaza request-ul).
func (c *Cache) Set(ctx context.Context, requestBody []byte, entry *Entry) {
    // L1: sync
    if err := c.exact.Set(ctx, requestBody, entry); err != nil {
        slog.Warn("cache L1 set failed", "err", err)
    }

    // L2: async — nu blocheaza response-ul catre client
    go func() {
        if err := c.semantic.Set(context.Background(), requestBody, entry); err != nil {
            slog.Warn("cache L2 set failed", "err", err)
        }
    }()
}

// Stats returns current cache statistics.
func (c *Cache) Stats(ctx context.Context) CacheStats { ... }

// CacheStats holds cache metrics.
type CacheStats struct {
    L1Entries int64
    L2Entries int64
    L1Hits    int64
    L2Hits    int64
    Misses    int64
}
```

## `exact.go` — L1 Cache (SHA256)

```go
package cache

import (
    "context"
    "crypto/sha256"
    "database/sql"
    "encoding/hex"
    "fmt"
    "time"
)

// ExactCache implements L1 cache using SHA256 key matching.
type ExactCache struct {
    db     *sql.DB
    config Config
}

func newExactCache(config Config) (*ExactCache, error) {
    db := config.DB.(*sql.DB)

    _, err := db.Exec(`
        CREATE TABLE IF NOT EXISTS cache_exact (
            key        TEXT    PRIMARY KEY,
            entry      BLOB    NOT NULL,
            created_at INTEGER NOT NULL,
            expires_at INTEGER NOT NULL,
            hit_count  INTEGER NOT NULL DEFAULT 0
        );
        CREATE INDEX IF NOT EXISTS idx_cache_exact_expires
            ON cache_exact(expires_at);
    `)
    if err != nil {
        return nil, fmt.Errorf("exact cache: create table: %w", err)
    }

    return &ExactCache{db: db, config: config}, nil
}

// key derivation: SHA256 de pe body normalizat
// Normalizam pentru a evita cache miss-uri din cauza whitespace sau field order
func cacheKey(requestBody []byte) string {
    // Normalizare simpla in v1: trim whitespace
    // In v2: sort JSON keys pentru a fi order-independent
    normalized := bytes.TrimSpace(requestBody)
    hash := sha256.Sum256(normalized)
    return hex.EncodeToString(hash[:])
}

// Get returns cached entry for requestBody, or (nil, false) on miss/expired.
func (c *ExactCache) Get(ctx context.Context, requestBody []byte) (*Entry, bool) {
    key := cacheKey(requestBody)
    now := time.Now().Unix()

    var entryBytes []byte
    err := c.db.QueryRowContext(ctx, `
        SELECT entry FROM cache_exact
        WHERE key = ? AND expires_at > ?
    `, key, now).Scan(&entryBytes)

    if err == sql.ErrNoRows {
        return nil, false
    }
    if err != nil {
        slog.Warn("exact cache get error", "err", err)
        return nil, false
    }

    // Incrementeaza hit count async
    go func() {
        _, _ = c.db.Exec(
            `UPDATE cache_exact SET hit_count = hit_count + 1 WHERE key = ?`, key,
        )
    }()

    entry, err := deserialize(entryBytes)
    if err != nil {
        slog.Warn("exact cache deserialize error", "err", err)
        return nil, false
    }

    return entry, true
}

// Set stores entry with TTL.
func (c *ExactCache) Set(ctx context.Context, requestBody []byte, entry *Entry) error {
    key := cacheKey(requestBody)
    entry.RequestHash = key

    data, err := entry.serialize()
    if err != nil {
        return fmt.Errorf("exact cache serialize: %w", err)
    }

    now := time.Now()
    expiresAt := now.Add(c.config.TTL).Unix()

    _, err = c.db.ExecContext(ctx, `
        INSERT INTO cache_exact (key, entry, created_at, expires_at)
        VALUES (?, ?, ?, ?)
        ON CONFLICT(key) DO UPDATE SET
            entry = excluded.entry,
            created_at = excluded.created_at,
            expires_at = excluded.expires_at,
            hit_count = 0
    `, key, data, now.Unix(), expiresAt)

    return err
}

// Evict removes expired entries. Apelat periodic din background goroutine.
func (c *ExactCache) Evict(ctx context.Context) (int64, error) {
    res, err := c.db.ExecContext(ctx,
        `DELETE FROM cache_exact WHERE expires_at <= ?`, time.Now().Unix(),
    )
    if err != nil {
        return 0, err
    }
    return res.RowsAffected()
}

// LRUEvict removes oldest entries when MaxEntries is exceeded.
func (c *ExactCache) LRUEvict(ctx context.Context) error {
    _, err := c.db.ExecContext(ctx, `
        DELETE FROM cache_exact WHERE key IN (
            SELECT key FROM cache_exact
            ORDER BY created_at ASC
            LIMIT MAX(0, (SELECT COUNT(*) FROM cache_exact) - ?)
        )
    `, c.config.MaxEntries)
    return err
}
```

## `embeddings.go` — Embedding Provider

L2 cache are nevoie de embeddings pentru a calcula similaritatea.  
Suportam **doua providere** configurabile, cu fallback clar.

```go
package cache

import (
    "bytes"
    "context"
    "encoding/json"
    "fmt"
    "net/http"
)

// EmbeddingProvider defines how to generate vector embeddings.
type EmbeddingProvider interface {
    Embed(ctx context.Context, text string) ([]float32, error)
    Dimensions() int
}

// --- Provider 1: OpenAI-compatible embeddings API ---

// OpenAIEmbedding calls /v1/embeddings endpoint.
// Functioneaza cu OpenAI, Azure OpenAI, Ollama, si orice OpenAI-compatible server.
type OpenAIEmbedding struct {
    BaseURL string // ex: "https://api.openai.com" sau "http://localhost:11434"
    APIKey  string // din env var, nu hardcodat
    Model   string // default: "text-embedding-3-small"
    dims    int    // 1536 pentru text-embedding-3-small
}

func NewOpenAIEmbedding(baseURL, apiKey, model string) *OpenAIEmbedding {
    if model == "" {
        model = "text-embedding-3-small"
    }
    return &OpenAIEmbedding{
        BaseURL: baseURL,
        APIKey:  apiKey,
        Model:   model,
        dims:    1536,
    }
}

func (e *OpenAIEmbedding) Embed(ctx context.Context, text string) ([]float32, error) {
    payload := map[string]interface{}{
        "model": e.Model,
        "input": text,
    }
    body, _ := json.Marshal(payload)

    req, err := http.NewRequestWithContext(ctx, "POST",
        e.BaseURL+"/v1/embeddings",
        bytes.NewReader(body),
    )
    if err != nil {
        return nil, err
    }
    req.Header.Set("Content-Type", "application/json")
    req.Header.Set("Authorization", "Bearer "+e.APIKey) // SECURITATE: din env, nu logat

    resp, err := http.DefaultClient.Do(req)
    if err != nil {
        return nil, fmt.Errorf("embedding request: %w", err)
    }
    defer resp.Body.Close()

    var result struct {
        Data []struct {
            Embedding []float32 `json:"embedding"`
        } `json:"data"`
    }
    if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
        return nil, fmt.Errorf("embedding decode: %w", err)
    }
    if len(result.Data) == 0 {
        return nil, fmt.Errorf("empty embedding response")
    }

    return result.Data[0].Embedding, nil
}

func (e *OpenAIEmbedding) Dimensions() int { return e.dims }

// --- Provider 2: Hash-based pseudo-embeddings (zero dependencies, v1 fallback) ---

// HashEmbedding generates deterministic pseudo-embeddings using SHA256.
// Nu e semantic real — e util doar pentru teste si dev local fara API key.
// NU se foloseste in productie pentru L2 cache.
type HashEmbedding struct{}

func (h *HashEmbedding) Embed(_ context.Context, text string) ([]float32, error) {
    hash := sha256.Sum256([]byte(text))
    dims := h.Dimensions()
    vec := make([]float32, dims)
    for i := 0; i < dims; i++ {
        vec[i] = float32(hash[i%32]) / 255.0
    }
    // Normalizeaza la unit vector
    return normalize(vec), nil
}

func (h *HashEmbedding) Dimensions() int { return 128 }

// normalize divides vector by its L2 norm.
func normalize(v []float32) []float32 {
    var sum float64
    for _, x := range v {
        sum += float64(x) * float64(x)
    }
    norm := float32(math.Sqrt(sum))
    if norm == 0 {
        return v
    }
    out := make([]float32, len(v))
    for i, x := range v {
        out[i] = x / norm
    }
    return out
}
```

## `semantic.go` — L2 Cache (sqlite-vec)

```go
package cache

import (
    "context"
    "database/sql"
    "encoding/binary"
    "fmt"
    "log/slog"
    "math"
    "time"
)

// SemanticCache implements L2 cache using cosine similarity via sqlite-vec.
type SemanticCache struct {
    db       *sql.DB
    config   Config
    embedder EmbeddingProvider
}

func newSemanticCache(config Config) (*SemanticCache, error) {
    // Daca nu exista embedding provider, L2 e disabled
    if config.EmbeddingProvider == nil {
        return &SemanticCache{config: config, embedder: nil}, nil
    }

    db := config.DB.(*sql.DB)
    dims := config.EmbeddingProvider.Dimensions()

    // Creeaza virtual table sqlite-vec pentru vector similarity search
    _, err := db.Exec(fmt.Sprintf(`
        CREATE VIRTUAL TABLE IF NOT EXISTS cache_semantic_vec
        USING vec0(embedding float[%d]);
    `, dims))
    if err != nil {
        return nil, fmt.Errorf("semantic cache: create vec table: %w", err)
    }

    // Tabel companion pentru metadata (entry bytes, timestamps)
    _, err = db.Exec(`
        CREATE TABLE IF NOT EXISTS cache_semantic_meta (
            rowid      INTEGER PRIMARY KEY,
            entry      BLOB    NOT NULL,
            text_hash  TEXT    NOT NULL,
            created_at INTEGER NOT NULL,
            expires_at INTEGER NOT NULL,
            hit_count  INTEGER NOT NULL DEFAULT 0
        );
        CREATE INDEX IF NOT EXISTS idx_semantic_expires
            ON cache_semantic_meta(expires_at);
    `)
    if err != nil {
        return nil, fmt.Errorf("semantic cache: create meta table: %w", err)
    }

    return &SemanticCache{db: db, config: config, embedder: config.EmbeddingProvider}, nil
}

// disabled returns true daca L2 e dezactivat (fara embedding provider).
func (s *SemanticCache) disabled() bool {
    return s.embedder == nil
}

// Get finds the most similar cached entry above the similarity threshold.
func (s *SemanticCache) Get(ctx context.Context, requestBody []byte) (*Entry, float64, bool) {
    if s.disabled() {
        return nil, 0, false
    }

    // Genereaza embedding pentru request-ul curent
    text := string(requestBody)
    vec, err := s.embedder.Embed(ctx, text)
    if err != nil {
        slog.Warn("L2 cache: embed failed", "err", err)
        return nil, 0, false
    }

    vecBytes := float32SliceToBytes(vec)
    now := time.Now().Unix()

    // KNN search — cel mai apropiat vecin din spatiul vectorial
    // sqlite-vec calculeaza distanta euclidiana; convertim la cosine similarity
    var rowid int64
    var distance float64

    err = s.db.QueryRowContext(ctx, `
        SELECT m.rowid, v.distance
        FROM cache_semantic_vec v
        JOIN cache_semantic_meta m ON m.rowid = v.rowid
        WHERE v.embedding MATCH ?
          AND m.expires_at > ?
          AND k = 1
        ORDER BY v.distance
    `, vecBytes, now).Scan(&rowid, &distance)

    if err == sql.ErrNoRows {
        return nil, 0, false
    }
    if err != nil {
        slog.Warn("L2 cache: knn search failed", "err", err)
        return nil, 0, false
    }

    // Converteste distanta euclidiana la cosine similarity
    // Pentru vectori normalizati: cosine = 1 - (distance^2 / 2)
    similarity := 1.0 - (distance*distance)/2.0

    if similarity < s.config.SemanticThreshold {
        slog.Debug("L2 cache: below threshold", "similarity", similarity, "threshold", s.config.SemanticThreshold)
        return nil, similarity, false
    }

    // Fetch entry
    var entryBytes []byte
    err = s.db.QueryRowContext(ctx,
        `SELECT entry FROM cache_semantic_meta WHERE rowid = ?`, rowid,
    ).Scan(&entryBytes)
    if err != nil {
        return nil, 0, false
    }

    go func() {
        _, _ = s.db.Exec(
            `UPDATE cache_semantic_meta SET hit_count = hit_count + 1 WHERE rowid = ?`, rowid,
        )
    }()

    entry, err := deserialize(entryBytes)
    if err != nil {
        return nil, 0, false
    }

    return entry, similarity, true
}

// Set generates embedding and stores entry in both vec and meta tables.
func (s *SemanticCache) Set(ctx context.Context, requestBody []byte, entry *Entry) error {
    if s.disabled() {
        return nil
    }

    vec, err := s.embedder.Embed(ctx, string(requestBody))
    if err != nil {
        return fmt.Errorf("L2 cache embed: %w", err)
    }

    data, err := entry.serialize()
    if err != nil {
        return fmt.Errorf("L2 cache serialize: %w", err)
    }

    vecBytes := float32SliceToBytes(vec)
    now := time.Now()
    expiresAt := now.Add(s.config.TTL).Unix()
    textHash := cacheKey(requestBody)

    // Atomic insert in ambele tabele via transaction
    tx, err := s.db.BeginTx(ctx, nil)
    if err != nil {
        return err
    }
    defer tx.Rollback()

    // Insert metadata si obtine rowid
    res, err := tx.ExecContext(ctx, `
        INSERT INTO cache_semantic_meta (entry, text_hash, created_at, expires_at)
        VALUES (?, ?, ?, ?)
    `, data, textHash, now.Unix(), expiresAt)
    if err != nil {
        return fmt.Errorf("L2 cache: insert meta: %w", err)
    }

    rowid, _ := res.LastInsertId()

    // Insert vector cu acelasi rowid
    _, err = tx.ExecContext(ctx, `
        INSERT INTO cache_semantic_vec (rowid, embedding)
        VALUES (?, ?)
    `, rowid, vecBytes)
    if err != nil {
        return fmt.Errorf("L2 cache: insert vec: %w", err)
    }

    return tx.Commit()
}

// float32SliceToBytes converts []float32 to []byte (little-endian) pentru sqlite-vec.
func float32SliceToBytes(v []float32) []byte {
    buf := make([]byte, len(v)*4)
    for i, f := range v {
        binary.LittleEndian.PutUint32(buf[i*4:], math.Float32bits(f))
    }
    return buf
}
```

## Eviction background goroutine

```go
func (c *Cache) startEviction(ctx context.Context) {
    ticker := time.NewTicker(1 * time.Hour)
    go func() {
        defer ticker.Stop()
        for {
            select {
            case <-ctx.Done():
                return
            case <-ticker.C:
                n, err := c.exact.Evict(ctx)
                if err != nil {
                    slog.Warn("eviction error", "err", err)
                } else if n > 0 {
                    slog.Debug("evicted expired entries", "count", n)
                }
                _ = c.exact.LRUEvict(ctx)
                // TODO SPEC-04: evict semantic entries
            }
        }
    }()
}

func TestExactCache_HitOnSameBody(t *testing.T) {
    db := openTestDB(t)
    c, _ := newExactCache(Config{DB: db, TTL: time.Hour, MaxEntries: 100})

    body := []byte(`{"model":"claude-3","messages":[{"role":"user","content":"hello"}]}`)
    entry := &Entry{StatusCode: 200, Body: []byte(`{"response":"world"}`)}

    _ = c.Set(context.Background(), body, entry)

    if !ok { t.Fatal("expected hit") }

    }
}

### Test 4: sanitizeHeaders nu lasa Authorization

func TestSanitizeHeaders_RemovesSensitive(t *testing.T) {
    h := http.Header{}
    h.Set("Authorization", "Bearer sk-secret")
    h.Set("X-Api-Key", "key123")
    h.Set("Content-Type", "application/json")

    sanitized := sanitizeHeaders(h)

    if sanitized.Get("Authorization") != "" { t.Error("Authorization should be removed") }
    if sanitized.Get("X-Api-Key") != "" { t.Error("X-Api-Key should be removed") }
    if sanitized.Get("Content-Type") == "" { t.Error("Content-Type should be kept") }
}
```
