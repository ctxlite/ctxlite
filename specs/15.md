## `.github/workflows/ci.yml`

```yaml
name: CI

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  test:
    name: Test & Typecheck
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: "20"
          cache: "npm"

      - name: Install dependencies
        run: npm ci

      - name: Typecheck all packages
        run: npm run typecheck

      - name: Build all packages
        run: npm run build

      - name: Run tests
        run: npm test

  validate-npm:
    name: Validate npm packages
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: "20"

      - name: Validate package.json files
        run: |
          for pkg in packages/core packages/opencode packages/mcp packages/cli; do
            echo "Checking $pkg/package.json..."
            node -e "require('./$pkg/package.json')"
          done

      - name: Check versions are in sync
        run: |
          ROOT_VERSION=$(node -e "console.log(require('./package.json').config.version)")
          for pkg in packages/core packages/opencode packages/mcp packages/cli; do
            PKG_VERSION=$(node -e "console.log(require('./$pkg/package.json').version)")
            if [ "$ROOT_VERSION" != "$PKG_VERSION" ]; then
              echo "Version mismatch in $pkg: $PKG_VERSION != $ROOT_VERSION"
              exit 1
            fi
          done
          echo "All versions in sync: $ROOT_VERSION"

      - name: Check no console.log in MCP server
        run: |
          if grep -r "console\.log" packages/mcp/src/; then
            echo "ERROR: console.log found in MCP server — use console.error instead"
            exit 1
          fi
          echo "OK: no console.log in MCP server"
```

## `.github/workflows/release.yml`

```yaml
name: Release

on:
  push:
    tags:
      - "v*.*.*"

permissions:
  contents: write

jobs:
  release:
    name: Publish to npm
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: "20"
          registry-url: "https://registry.npmjs.org"
          cache: "npm"

      - name: Install dependencies
        run: npm ci

      - name: Build all packages
        run: npm run build

      - name: Run tests
        run: npm test

      - name: Extract version from tag
        id: version
        run: echo "VERSION=${GITHUB_REF_NAME#v}" >> $GITHUB_OUTPUT

      - name: Verify version matches tag
        run: |
          ROOT_VERSION=$(node -e "console.log(require('./package.json').config.version)")
          TAG_VERSION="${{ steps.version.outputs.VERSION }}"
          if [ "$ROOT_VERSION" != "$TAG_VERSION" ]; then
            echo "ERROR: package.json version ($ROOT_VERSION) != tag ($TAG_VERSION)"
            echo "Run: node scripts/sync-version.js after updating config.version"
            exit 1
          fi

      # Publica in ordinea corecta — core inainte de pachete care depind de el
      - name: Publish @ctxlite/core
        env:
          NODE_AUTH_TOKEN: ${{ secrets.NPM_TOKEN }}
        run: cd packages/core && npm publish --access public

      - name: Publish @ctxlite/opencode
        env:
          NODE_AUTH_TOKEN: ${{ secrets.NPM_TOKEN }}
        run: cd packages/opencode && npm publish --access public

      - name: Publish @ctxlite/mcp
        env:
          NODE_AUTH_TOKEN: ${{ secrets.NPM_TOKEN }}
        run: cd packages/mcp && npm publish --access public

      - name: Publish ctxlite (CLI)
        env:
          NODE_AUTH_TOKEN: ${{ secrets.NPM_TOKEN }}
        run: cd packages/cli && npm publish --access public

      - name: Create GitHub Release
        uses: softprops/action-gh-release@v2
        with:
          name: "ctxlite ${{ github.ref_name }}"
          generate_release_notes: true
          prerelease: ${{ contains(github.ref_name, '-') }}
          body: |
            ## Install

            **OpenCode:**
            ```bash
            opencode plugin @ctxlite/opencode -g -f
            ```

            **Cursor:**
            ```json
            {
              "mcpServers": {
                "ctxlite": { "command": "npx", "args": ["-y", "@ctxlite/mcp"] }
              }
            }
            ```

            **CLI stats:**
            ```bash
            npx ctxlite stats
            ```
```

## Release flow (manual, pentru maintainer)

```bash
# 1. Actualizeaza versiunea in root package.json
# Editeaza: "config": { "version": "0.2.0" }

# 2. Sincronizeaza in toate pachetele
node scripts/sync-version.js

# 3. Actualizeaza CHANGELOG.md

# 4. Commit + tag
git add -A
git commit -m "chore: release v0.2.0"
git tag v0.2.0
git push origin main --tags

# CI publica automat pe npm si creeaza GitHub Release
```

## Secret necesar

```
NPM_TOKEN  ← npm automation token
```

Adauga in GitHub repo → Settings → Secrets → Actions.

Genereaza cu: `npm token create --type=automation`
