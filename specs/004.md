## Ce face trimmer-ul

```
Request body (JSON)
        ↓
[1] Parseaza mesajele → extrage query-ul userului + fisierele din context
        ↓
[2] BM25 scoring → scoreaza fiecare fisier fata de query
        ↓
[3] Import graph boost → fisierele importate de fisierele top-scored primesc bonus
        ↓
[4] Selectie → pastreaza fisierele peste threshold pana la budget de tokens
        ↓
[5] Reconstruieste request body → inlocuieste continutul fisierelor cu cele trimmed
        ↓
Trimmed request body (JSON) + TokensSaved int
```

```go
package trimmer

// ParsedRequest holds extracted data from an API request body.
type ParsedRequest struct {
    Format   APIFormat   // Anthropic sau OpenAI
    Query    string      // ultimul mesaj user, fara code blocks
    Files    []CodeFile  // fisierele detectate in context
    Raw      []byte      // body-ul original complet
}

// APIFormat identifies the upstream API format.
type APIFormat int

const (
    FormatUnknown   APIFormat = iota
    FormatAnthropic           // /v1/messages
    FormatOpenAI              // /v1/chat/completions
)

// CodeFile represents a file extracted from the request context.
type CodeFile struct {
    Path     string // ex: "internal/auth/login.go"
    Content  string // continutul fisierului
    Language string // "go", "python", "typescript", etc.
    Tokens   int    // token count estimat
}
```

### Implementare

```go
// Parse extracts query and files from an API request body.
// Returns ParsedRequest with empty Files if no code blocks are found.
func Parse(body []byte) (ParsedRequest, error) {
    // Detecteaza formatul din structura JSON
    var probe struct {
        Messages []json.RawMessage `json:"messages"`
    }
    if err := json.Unmarshal(body, &probe); err != nil {
        return ParsedRequest{Raw: body}, fmt.Errorf("parse: unmarshal: %w", err)
    }

    format := detectFormat(body)

    switch format {
    case FormatAnthropic:
        return parseAnthropic(body)
    case FormatOpenAI:
        return parseOpenAI(body)
    default:
        return ParsedRequest{Raw: body, Format: FormatUnknown}, nil
    }
}

// detectFormat returns APIFormat based on JSON structure.
func detectFormat(body []byte) APIFormat {
    // Anthropic are "content" ca array de objects cu "type"
    // OpenAI are "content" ca string sau array simplu
    if bytes.Contains(body, []byte(`"anthropic-version"`)) ||
       bytes.Contains(body, []byte(`"type":"text"`) ) {
        // Verificare mai riguroasa: primul mesaj are content array cu type field
        return FormatAnthropic
    }
    if bytes.Contains(body, []byte(`"chat/completions"`) ) ||
       bytes.Contains(body, []byte(`"role":"system"`)) {
        return FormatOpenAI
    }
    // Fallback: incearca sa parseze ca Anthropic
    return FormatAnthropic
}

// extractCodeBlocks gaseste fisierele in format:
// ```lang\n// File: path\ncontent\n```
// sau
// ```lang\n// path\ncontent\n```
var codeBlockRe = regexp.MustCompile("(?s)```(\\w*)\\n(?://\\s*(?:File:\\s*)?([^\\n]+)\\n)?(.*?)```")

func extractCodeBlocks(text string) []CodeFile {
    matches := codeBlockRe.FindAllStringSubmatch(text, -1)
    var files []CodeFile
    for _, m := range matches {
        lang := m[1]
        path := strings.TrimSpace(m[2])
        content := m[3]

        if path == "" {
            path = fmt.Sprintf("snippet_%d.%s", len(files), lang)
        }

        files = append(files, CodeFile{
            Path:     path,
            Content:  content,
            Language: lang,
            Tokens:   len(content) / 4, // estimare, actualizat in SPEC-04
        })
    }
    return files
}

// extractQuery returneaza textul userului fara code blocks.
func extractQuery(text string) string {
    clean := codeBlockRe.ReplaceAllString(text, "")
    return strings.TrimSpace(clean)
}
```

## `bm25.go` — BM25 Scoring

BM25 (Best Match 25) este algoritmul standard pentru information retrieval.  
Nu necesita embeddings sau modele ML — e pur matematic.

```go
package trimmer

import (
    "math"
    "regexp"
    "strings"
)

// BM25Params holds BM25 hyperparameters.
type BM25Params struct {
    K1 float64 // term frequency saturation (default: 1.5)
    B  float64 // length normalization (default: 0.75)
}

// DefaultBM25Params returns sensible defaults for code files.
var DefaultBM25Params = BM25Params{K1: 1.5, B: 0.75}

// BM25 scores documents against a query using BM25 algorithm.
type BM25 struct {
    params    BM25Params
    docs      [][]string // tokenized documents
    avgDocLen float64
    idf       map[string]float64
}

// NewBM25 creates a BM25 scorer for the given documents.
// docs[i] corresponds to files[i].
func NewBM25(docs []string, params BM25Params) *BM25 {
    tokenized := make([][]string, len(docs))
    totalLen := 0
    for i, doc := range docs {
        tokens := tokenize(doc)
        tokenized[i] = tokens
        totalLen += len(tokens)
    }

    avgLen := 0.0
    if len(docs) > 0 {
        avgLen = float64(totalLen) / float64(len(docs))
    }

    return &BM25{
        params:    params,
        docs:      tokenized,
        avgDocLen: avgLen,
        idf:       computeIDF(tokenized),
    }
}

// Score returns BM25 relevance score for docIndex against query.
func (b *BM25) Score(docIndex int, query string) float64 {
    queryTokens := tokenize(query)
    doc := b.docs[docIndex]
    docLen := float64(len(doc))

    // Frecventa fiecarui token in document
    tf := make(map[string]int)
    for _, t := range doc {
        tf[t]++
    }

    score := 0.0
    for _, qt := range queryTokens {
        idf := b.idf[qt]
        freq := float64(tf[qt])

        // BM25 formula
        numerator := freq * (b.params.K1 + 1)
        denominator := freq + b.params.K1*(1-b.params.B+b.params.B*(docLen/b.avgDocLen))
        score += idf * (numerator / denominator)
    }
    return score
}

// ScoreAll returns scores for all documents, sorted descending.
func (b *BM25) ScoreAll(query string) []ScoredDoc {
    scores := make([]ScoredDoc, len(b.docs))
    for i := range b.docs {
        scores[i] = ScoredDoc{Index: i, Score: b.Score(i, query)}
    }
    sort.Slice(scores, func(i, j int) bool {
        return scores[i].Score > scores[j].Score
    })
    return scores
}

// ScoredDoc pairs a document index with its BM25 score.
type ScoredDoc struct {
    Index int
    Score float64
}

// computeIDF calculates inverse document frequency for all terms.
func computeIDF(docs [][]string) map[string]float64 {
    N := float64(len(docs))
    df := make(map[string]int)
    for _, doc := range docs {
        seen := make(map[string]bool)
        for _, t := range doc {
            if !seen[t] {
                df[t]++
                seen[t] = true
            }
        }
    }

    idf := make(map[string]float64)
    for term, freq := range df {
        // IDF cu smoothing
        idf[term] = math.Log((N-float64(freq)+0.5)/(float64(freq)+0.5) + 1)
    }
    return idf
}

// tokenize splits text into lowercase tokens.
// Optimizat pentru cod sursa: split pe camelCase, snake_case, spatii, punctuatie.
var tokenRe = regexp.MustCompile(`[A-Z][a-z]+|[a-z]+|[A-Z]+(?:[A-Z][a-z]+)*|\d+`)

func tokenize(text string) []string {
    // Lowercase tot
    lower := strings.ToLower(text)
    // Inlocuieste separatori comuni cu spatiu
    lower = strings.NewReplacer(
        "_", " ", "-", " ", ".", " ", "/", " ", "(", " ", ")", " ",
        "{", " ", "}", " ", "[", " ", "]", " ", ";", " ", ":", " ",
    ).Replace(lower)

    tokens := strings.Fields(lower)

    // Filtreaza stop words si tokens prea scurte
    filtered := tokens[:0]
    for _, t := range tokens {
        if len(t) >= 2 && !isStopWord(t) {
            filtered = append(filtered, t)
        }
    }
    return filtered
}

var stopWords = map[string]bool{
    "the": true, "a": true, "an": true, "is": true, "it": true,
    "in": true, "on": true, "at": true, "to": true, "for": true,
    "of": true, "and": true, "or": true, "but": true, "not": true,
    "with": true, "this": true, "that": true, "be": true, "as": true,
    // Code-specific stop words
    "return": true, "func": true, "var": true, "const": true,
    "if": true, "else": true, "for": true, "import": true,
}

func isStopWord(t string) bool { return stopWords[t] }
```

## `imports.go` — Import Graph cu tree-sitter

Tree-sitter analizeaza AST-ul fiecarui fisier si extrage import-urile.  
Fisierele importate de fisierele relevante primesc un boost de relevanta.

```go
package trimmer

import (
    "path/filepath"
    "strings"

    sitter "github.com/smacker/go-tree-sitter"
    "github.com/smacker/go-tree-sitter/golang"
    "github.com/smacker/go-tree-sitter/python"
    "github.com/smacker/go-tree-sitter/typescript/typescript"
    "github.com/smacker/go-tree-sitter/javascript"
)

// ImportGraph maps file paths to the files they import.
type ImportGraph map[string][]string

// BuildImportGraph analyzes all files and returns their import relationships.
func BuildImportGraph(files []CodeFile) ImportGraph {
    graph := make(ImportGraph)
    for _, f := range files {
        imports := extractImports(f)
        if len(imports) > 0 {
            graph[f.Path] = imports
        }
    }
    return graph
}

// extractImports uses tree-sitter to extract import paths from a file.
func extractImports(file CodeFile) []string {
    lang := detectLanguage(file)
    if lang == nil {
        return nil
    }

    parser := sitter.NewParser()
    parser.SetLanguage(lang)

    tree, err := parser.ParseCtx(context.Background(), nil, []byte(file.Content))
    if err != nil || tree == nil {
        return nil
    }

    switch file.Language {
    case "go":
        return extractGoImports(tree, file.Content)
    case "python":
        return extractPythonImports(tree, file.Content)
    case "typescript", "javascript":
        return extractJSImports(tree, file.Content)
    }
    return nil
}

// detectLanguage returns the tree-sitter language for a file.
func detectLanguage(file CodeFile) *sitter.Language {
    switch strings.ToLower(file.Language) {
    case "go":
        return golang.GetLanguage()
    case "python", "py":
        return python.GetLanguage()
    case "typescript", "ts":
        return typescript.GetLanguage()
    case "javascript", "js":
        return javascript.GetLanguage()
    }

    // Fallback din extensia path-ului
    switch filepath.Ext(file.Path) {
    case ".go":
        return golang.GetLanguage()
    case ".py":
        return python.GetLanguage()
    case ".ts":
        return typescript.GetLanguage()
    case ".js":
        return javascript.GetLanguage()
    }
    return nil
}

// extractGoImports extracts import paths from a Go file AST.
// Returneaza doar import-urile relative (incep cu "./" sau "../").
func extractGoImports(tree *sitter.Tree, content string) []string {
    // Query tree-sitter pentru import declarations
    // node type: import_spec → path → string content
    var imports []string
    src := []byte(content)

    var walk func(node *sitter.Node)
    walk = func(node *sitter.Node) {
        if node.Type() == "interpreted_string_literal" {
            parent := node.Parent()
            if parent != nil && parent.Type() == "import_spec" {
                path := strings.Trim(string(node.Content(src)), `"`)
                // Pastreaza doar import-urile relative sau din modul curent
                if strings.HasPrefix(path, "./") || strings.HasPrefix(path, "../") {
                    imports = append(imports, path)
                }
            }
        }
        for i := 0; i < int(node.ChildCount()); i++ {
            walk(node.Child(i))
        }
    }
    walk(tree.RootNode())
    return imports
}

// extractPythonImports extrage import-urile relative din Python.
func extractPythonImports(tree *sitter.Tree, content string) []string {
    var imports []string
    src := []byte(content)

    var walk func(node *sitter.Node)
    walk = func(node *sitter.Node) {
        // relative_import: from . import X sau from .module import X
        if node.Type() == "relative_import" {
            imports = append(imports, string(node.Content(src)))
        }
        for i := 0; i < int(node.ChildCount()); i++ {
            walk(node.Child(i))
        }
    }
    walk(tree.RootNode())
    return imports
}

// extractJSImports extrage import-urile relative din JS/TS.
func extractJSImports(tree *sitter.Tree, content string) []string {
    var imports []string
    src := []byte(content)

    var walk func(node *sitter.Node)
    walk = func(node *sitter.Node) {
        if node.Type() == "import_statement" {
            for i := 0; i < int(node.ChildCount()); i++ {
                child := node.Child(i)
                if child.Type() == "string" {
                    path := strings.Trim(string(child.Content(src)), `"'`)
                    if strings.HasPrefix(path, ".") {
                        imports = append(imports, path)
                    }
                }
            }
        }
        for i := 0; i < int(node.ChildCount()); i++ {
            walk(node.Child(i))
        }
    }
    walk(tree.RootNode())
    return imports
}

// ImportBoost returns a boost score for files imported by top-scored files.
// Boost = 0.3 * (score al fisierului care face import) pentru fiecare import relationship.
func ImportBoost(graph ImportGraph, scores map[string]float64) map[string]float64 {
    boosts := make(map[string]float64)
    const boostFactor = 0.3

    for importer, imported := range graph {
        importerScore := scores[importer]
        if importerScore <= 0 {
            continue
        }
        for _, dep := range imported {
            boosts[dep] += importerScore * boostFactor
        }
    }
    return boosts
}
```

## `budget.go` — Selectie fisiere in limita de token budget

```go
package trimmer

import "sort"

// SelectionResult holds the output of file selection.
type SelectionResult struct {
    Selected    []CodeFile // fisierele selectate, in ordinea relevantei
    TokensUsed  int        // total tokens in fisierele selectate
    TokensSaved int        // tokens eliminati
    TrimRatio   float64    // (1 - TokensUsed/TotalTokens)
}

// SelectFiles picks the most relevant files within the token budget.
func SelectFiles(
    files []CodeFile,
    scores []ScoredDoc,     // BM25 scores, sorted descending
    importBoosts map[string]float64,
    maxTokens int,           // default: 4096 tokens pentru context
    minScore float64,        // default: 0.1 (filtreaza fisierele complet nerelevante)
) SelectionResult {
    // Calculeaza total tokens original
    totalTokens := 0
    for _, f := range files {
        totalTokens += f.Tokens
    }

    // Aplica import boost la scores
    fileScores := make(map[string]float64)
    for _, sd := range scores {
        fileScores[files[sd.Index].Path] = sd.Score
    }
    for path, boost := range importBoosts {
        fileScores[path] += boost
    }

    // Resorteza cu boost aplicat
    type scoredFile struct {
        file  CodeFile
        score float64
    }
    scored := make([]scoredFile, len(files))
    for i, f := range files {
        scored[i] = scoredFile{file: f, score: fileScores[f.Path]}
    }
    sort.Slice(scored, func(i, j int) bool {
        return scored[i].score > scored[j].score
    })

    // Selecteaza fisiere pana la budget
    var selected []CodeFile
    tokensUsed := 0

    for _, sf := range scored {
        // Skip fisiere sub threshold de relevanta
        if sf.score < minScore {
            continue
        }
        // Skip daca depaseste budget
        if tokensUsed+sf.file.Tokens > maxTokens {
            continue
        }
        selected = append(selected, sf.file)
        tokensUsed += sf.file.Tokens
    }

    tokensSaved := totalTokens - tokensUsed
    trimRatio := 0.0
    if totalTokens > 0 {
        trimRatio = float64(tokensSaved) / float64(totalTokens)
    }

    return SelectionResult{
        Selected:    selected,
        TokensUsed:  tokensUsed,
        TokensSaved: tokensSaved,
        TrimRatio:   trimRatio,
    }
}
```

## `tokens.go` — Token Counting Real din Response

Inlocuieste `estimateTokens()` placeholder din SPEC-02/03.

```go
package trimmer

import "encoding/json"

// UsageAnthropic matches Anthropic API response usage field.
type UsageAnthropic struct {
    InputTokens  int `json:"input_tokens"`
    OutputTokens int `json:"output_tokens"`
}

// UsageOpenAI matches OpenAI API response usage field.
type UsageOpenAI struct {
    PromptTokens     int `json:"prompt_tokens"`
    CompletionTokens int `json:"completion_tokens"`
    TotalTokens      int `json:"total_tokens"`
}

// TokenUsage holds normalized token counts from any API response.
type TokenUsage struct {
    InputTokens  int
    OutputTokens int
    TotalTokens  int
}

// ParseTokenUsage extracts token counts from an API response body.
// Suporta atat Anthropic cat si OpenAI format.
// Returneaza zero usage daca parsing-ul esueaza — nu crapa requestul.
func ParseTokenUsage(responseBody []byte) TokenUsage {
    // Incearca Anthropic format
    var anthropic struct {
        Usage UsageAnthropic `json:"usage"`
    }
    if err := json.Unmarshal(responseBody, &anthropic); err == nil {
        if anthropic.Usage.InputTokens > 0 || anthropic.Usage.OutputTokens > 0 {
            total := anthropic.Usage.InputTokens + anthropic.Usage.OutputTokens
            return TokenUsage{
                InputTokens:  anthropic.Usage.InputTokens,
                OutputTokens: anthropic.Usage.OutputTokens,
                TotalTokens:  total,
            }
        }
    }

    // Incearca OpenAI format
    var openai struct {
        Usage UsageOpenAI `json:"usage"`
    }
    if err := json.Unmarshal(responseBody, &openai); err == nil {
        if openai.Usage.TotalTokens > 0 {
            return TokenUsage{
                InputTokens:  openai.Usage.PromptTokens,
                OutputTokens: openai.Usage.CompletionTokens,
                TotalTokens:  openai.Usage.TotalTokens,
            }
        }
    }

    return TokenUsage{} // zero — nu blocheaza nimic
}

// ParseStreamTokenUsage extrage token usage din ultimul event SSE.
// Anthropic trimite usage in event-ul de tip "message_delta".
// OpenAI trimite usage in ultimul chunk cu finish_reason != null.
func ParseStreamTokenUsage(events []string) TokenUsage {
    // Cauta ultimul event JSON care contine "usage"
    for i := len(events) - 1; i >= 0; i-- {
        line := events[i]
        if !strings.HasPrefix(line, "data: ") {
            continue
        }
        data := strings.TrimPrefix(line, "data: ")
        if data == "[DONE]" {
            continue
        }

        usage := ParseTokenUsage([]byte(data))
        if usage.TotalTokens > 0 {
            return usage
        }
    }
    return TokenUsage{}
}

// EstimateTokens gives a rough estimate when no usage data is available.
// ~4 characters per token for English/code.
func EstimateTokens(text []byte) int {
    return len(text) / 4
}
```

## `trimmer.go` — Entry Point

```go
package trimmer

import (
    "fmt"
    "log/slog"
)

// Config holds trimmer configuration.
type Config struct {
    MaxContextTokens int     // default: 4096
    MinBM25Score     float64 // default: 0.1
    EnableImportGraph bool   // default: true
    BM25             BM25Params
}

// TrimResult holds the result of a trim operation.
type TrimResult struct {
    Body        []byte // request body modificat
    TokensIn    int    // tokens originali
    TokensUsed  int    // tokens dupa trim
    TokensSaved int    // diferenta
    TrimRatio   float64
    FilesIn     int
    FilesOut    int
}

// Trimmer performs context trimming on API request bodies.
type Trimmer struct {
    config Config
}

// New creates a Trimmer with the given config.
func New(config Config) *Trimmer {
    if config.MaxContextTokens == 0 {
        config.MaxContextTokens = 4096
    }
    if config.MinBM25Score == 0 {
        config.MinBM25Score = 0.1
    }
    if config.BM25.K1 == 0 {
        config.BM25 = DefaultBM25Params
    }
    config.EnableImportGraph = true // default on
    return &Trimmer{config: config}
}

// Trim analyzes and trims the request body.
// Returneaza body-ul original nemodificat daca:
//   - nu se gasesc fisiere in context
//   - query-ul e gol
//   - parsarea esueaza
func (t *Trimmer) Trim(body []byte) TrimResult {
    // Parse request
    parsed, err := Parse(body)
    if err != nil {
        slog.Debug("trimmer: parse failed, skipping", "err", err)
        return TrimResult{Body: body, TokensIn: EstimateTokens(body)}
    }

    // Nu trim daca nu sunt fisiere sau query
    if len(parsed.Files) == 0 || parsed.Query == "" {
        return TrimResult{
            Body:     body,
            TokensIn: EstimateTokens(body),
            FilesIn:  len(parsed.Files),
            FilesOut: len(parsed.Files),
        }
    }

    tokensIn := 0
    for _, f := range parsed.Files {
        tokensIn += f.Tokens
    }

    // BM25 scoring
    docs := make([]string, len(parsed.Files))
    for i, f := range parsed.Files {
        docs[i] = f.Path + " " + f.Content
    }
    scorer := NewBM25(docs, t.config.BM25)
    scores := scorer.ScoreAll(parsed.Query)

    // Import graph boost
    var importBoosts map[string]float64
    if t.config.EnableImportGraph {
        graph := BuildImportGraph(parsed.Files)
        scoreMap := make(map[string]float64)
        for _, sd := range scores {
            scoreMap[parsed.Files[sd.Index].Path] = sd.Score
        }
        importBoosts = ImportBoost(graph, scoreMap)
    }

    // Selectie in limita de budget
    selection := SelectFiles(
        parsed.Files,
        scores,
        importBoosts,
        t.config.MaxContextTokens,
        t.config.MinBM25Score,
    )

    // Daca trim-ul nu salveaza nimic semnificativ (< 10%), returneaza original
    if selection.TrimRatio < 0.10 {
        return TrimResult{
            Body:     body,
            TokensIn: tokensIn,
            FilesIn:  len(parsed.Files),
            FilesOut: len(parsed.Files),
        }
    }

    // Reconstruieste request body cu fisierele selectate
    trimmedBody, err := rebuildRequest(parsed, selection.Selected)
    if err != nil {
        slog.Warn("trimmer: rebuild failed, using original", "err", err)
        return TrimResult{Body: body, TokensIn: tokensIn}
    }

    slog.Debug("trimmer: trimmed",
        "files_in", len(parsed.Files),
        "files_out", len(selection.Selected),
        "tokens_in", tokensIn,
        "tokens_saved", selection.TokensSaved,
        "ratio", fmt.Sprintf("%.0f%%", selection.TrimRatio*100),
    )

    return TrimResult{
        Body:        trimmedBody,
        TokensIn:    tokensIn,
        TokensUsed:  selection.TokensUsed,
        TokensSaved: selection.TokensSaved,
        TrimRatio:   selection.TrimRatio,
        FilesIn:     len(parsed.Files),
        FilesOut:    len(selection.Selected),
    }
}

// rebuildRequest reconstructs the API request body with only selected files.
func rebuildRequest(parsed ParsedRequest, selected []CodeFile) ([]byte, error) {
    // Strategia: inlocuieste in body-ul original toate fisierele
    // cu doar cele din selected. Fisierele eliminate sunt sterse complet.
    // Structura JSON (model, system prompt, etc.) ramane intacta.

    // Pentru simplitate in v1: re-serializam mesajele cu fisierele selectate
    // Implementare detaliata depinde de format (Anthropic vs OpenAI)
    switch parsed.Format {
    case FormatAnthropic:
        return rebuildAnthropic(parsed, selected)
    case FormatOpenAI:
        return rebuildOpenAI(parsed, selected)
    default:
        return parsed.Raw, nil
    }
}
```

```go
    }
    return float64(tokensSaved) / 1_000_000 * price
```
