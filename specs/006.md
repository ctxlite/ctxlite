# SPEC-06: npm Wrapper Packages

**Project:** `ctxlite`  
**Scope:** Structura completa de npm packages pentru distributia binarului Go — wrapper principal + 5 pachete platform-specific  
**Depinde de:** SPEC-01 (structura repo), SPEC-07 (CI care populeaza binarele)  
**Agent target:** Cursor AI  
**Status:** `[x] Done`

## Context

Userul instaleaza `ctxlite` cu o singura comanda:

```bash
npm install -g ctxlite
# sau fara install global:
npx -y ctxlite
```

npm detecteaza automat OS + arch si descarca **doar** pachetul platform-specific relevant.  
La runtime, `bin/ctxlite.js` localizeaza binarul Go si il executa direct — zero overhead Node.js dupa start.

Pattern folosit de: `esbuild`, `@biomejs/biome`, `@tailwindcss/cli`, `prisma`.

## `npm/ctxlite/bin/ctxlite.js`

```js
#!/usr/bin/env node

"use strict";

const { execFileSync } = require("child_process");
const path = require("path");
const os = require("os");
const fs = require("fs");

// Map platform + arch la numele pachetului npm platform-specific
const PLATFORM_PACKAGES = {
  "darwin-arm64": "ctxlite-darwin-arm64",
  "darwin-x64":   "ctxlite-darwin-x64",
  "linux-x64":    "ctxlite-linux-x64",
  "linux-arm64":  "ctxlite-linux-arm64",
  "win32-x64":    "ctxlite-win32-x64",
};

function getPlatformKey() {
  const platform = os.platform(); // "darwin" | "linux" | "win32"
  const arch = os.arch();         // "x64" | "arm64"
  return `${platform}-${arch}`;
}

function getBinaryPath() {
  const key = getPlatformKey();
  const pkg = PLATFORM_PACKAGES[key];

  if (!pkg) {
    throw new Error(
      `ctxlite: unsupported platform ${key}\n` +
      `Supported: ${Object.keys(PLATFORM_PACKAGES).join(", ")}\n` +
      `Please open an issue: https://github.com/ctxlite/ctxlite/issues`
    );
  }

  // Localizeaza package-ul platform-specific instalat de npm
  let pkgJsonPath;
  try {
    pkgJsonPath = require.resolve(`${pkg}/package.json`);
  } catch (e) {
    throw new Error(
      `ctxlite: platform package ${pkg} not found.\n` +
      `Try reinstalling: npm install -g ctxlite\n` +
      `If the problem persists, install manually: npm install -g ${pkg}`
    );
  }

  const binName = os.platform() === "win32" ? "ctxlite.exe" : "ctxlite";
  const binPath = path.join(path.dirname(pkgJsonPath), "bin", binName);

  if (!fs.existsSync(binPath)) {
    throw new Error(
      `ctxlite: binary not found at ${binPath}\n` +
      `This may happen if the package was installed without binaries.\n` +
      `Try: npm install -g ctxlite`
    );
  }

  return binPath;
}

// Entry point
try {
  const bin = getBinaryPath();
  // Executa binarul Go cu toate argumentele originale
  // stdio: "inherit" = stdin/stdout/stderr conectate direct — critic pentru MCP stdio
  execFileSync(bin, process.argv.slice(2), {
    stdio: "inherit",
    windowsHide: false,
  });
} catch (err) {
  // ExecFileSync arunca eroare daca procesul iese cu exit code != 0
  // In acest caz, binarul a afisat deja eroarea — exit cu acelasi cod
  if (err.status !== undefined) {
    process.exit(err.status);
  }
  // Altfel e o eroare de setup — arata mesaj clar
  process.stderr.write(err.message + "\n");
  process.exit(1);
}
```

## `npm/ctxlite/scripts/install.js`

Script rulat automat de npm dupa `npm install -g ctxlite`.  
Verifica ca binarul e prezent si executabil. Esueaza **silentios** daca nu e — nu vrem sa bloam install-ul.

```js
"use strict";

const path = require("path");
const os = require("os");
const fs = require("fs");

const PLATFORM_PACKAGES = {
  "darwin-arm64": "ctxlite-darwin-arm64",
  "darwin-x64":   "ctxlite-darwin-x64",
  "linux-x64":    "ctxlite-linux-x64",
  "linux-arm64":  "ctxlite-linux-arm64",
  "win32-x64":    "ctxlite-win32-x64",
};

function main() {
  const key = `${os.platform()}-${os.arch()}`;
  const pkg = PLATFORM_PACKAGES[key];

  if (!pkg) {
    // Platform nesupotata — avertizare, nu eroare fatala
    process.stderr.write(
      `[ctxlite] Warning: platform ${key} is not supported.\n`
    );
    return;
  }

  let pkgJsonPath;
  try {
    pkgJsonPath = require.resolve(`${pkg}/package.json`);
  } catch {
    // Package optional nu s-a instalat — normal pe platforme nesuportate
    process.stderr.write(
      `[ctxlite] Warning: could not find ${pkg}. Try: npm install -g ctxlite\n`
    );
    return;
  }

  const binName = os.platform() === "win32" ? "ctxlite.exe" : "ctxlite";
  const binPath = path.join(path.dirname(pkgJsonPath), "bin", binName);

  if (!fs.existsSync(binPath)) {
    process.stderr.write(
      `[ctxlite] Warning: binary not found at ${binPath}\n`
    );
    return;
  }

  // Pe Unix, verifica ca e executabil
  if (os.platform() !== "win32") {
    try {
      fs.accessSync(binPath, fs.constants.X_OK);
    } catch {
      // Incearca sa seteze permisiunile
      try {
        fs.chmodSync(binPath, 0o755);
      } catch {
        process.stderr.write(
          `[ctxlite] Warning: binary at ${binPath} may not be executable.\n` +
          `Try: chmod +x ${binPath}\n`
        );
        return;
      }
    }
  }

  process.stdout.write(
    `[ctxlite] Binary ready: ${binPath}\n` +
    `[ctxlite] Run 'ctxlite --version' to verify installation.\n`
  );
}

main();
```

## Update `Makefile` — comenzi npm

Adauga in `Makefile`:

```makefile
# Versiunea curenta — sincronizata cu go/cmd/ctxlite/main.go
VERSION := 0.1.0

# Seteaza versiunea in toate package.json-urile
npm-version:
	node -e " \
	  const fs = require('fs'); \
	  const pkgs = [ \
	    'npm/ctxlite', \
	    'npm/ctxlite-darwin-arm64', \
	    'npm/ctxlite-darwin-x64', \
	    'npm/ctxlite-linux-x64', \
	    'npm/ctxlite-linux-arm64', \
	    'npm/ctxlite-win32-x64', \
	  ]; \
	  pkgs.forEach(p => { \
	    const f = p + '/package.json'; \
	    const j = JSON.parse(fs.readFileSync(f)); \
	    j.version = '$(VERSION)'; \
	    if (j.optionalDependencies) { \
	      Object.keys(j.optionalDependencies).forEach(k => j.optionalDependencies[k] = '$(VERSION)'); \
	    } \
	    fs.writeFileSync(f, JSON.stringify(j, null, 2) + '\n'); \
	    console.log('Updated', f); \
	  }); \
	"

# Test local: instaleaza binarul local in npm/ctxlite-*/bin/
npm-install-local:
	make build-all
	cp bin/ctxlite-darwin-arm64  npm/ctxlite-darwin-arm64/bin/ctxlite
	cp bin/ctxlite-darwin-x64    npm/ctxlite-darwin-x64/bin/ctxlite
	cp bin/ctxlite-linux-x64     npm/ctxlite-linux-x64/bin/ctxlite
	cp bin/ctxlite-linux-arm64   npm/ctxlite-linux-arm64/bin/ctxlite
	cp bin/ctxlite-win32-x64.exe npm/ctxlite-win32-x64/bin/ctxlite.exe
	chmod +x npm/ctxlite-darwin-arm64/bin/ctxlite
	chmod +x npm/ctxlite-darwin-x64/bin/ctxlite
	chmod +x npm/ctxlite-linux-x64/bin/ctxlite
	chmod +x npm/ctxlite-linux-arm64/bin/ctxlite

# Dry-run publish (verifica ce ar fi publicat fara a publica)
npm-pack:
	cd npm/ctxlite && npm pack --dry-run
	cd npm/ctxlite-darwin-arm64 && npm pack --dry-run
```

## Acceptance criteria

Agentul a terminat SPEC-06 cand:

- [ ] Toate cele 6 `package.json`-uri exista si sunt valid JSON (`node -e "require('./package.json')"`)
- [ ] `npm/ctxlite/bin/ctxlite.js` are shebang `#!/usr/bin/env node`
- [ ] `npm/ctxlite/bin/ctxlite.js` are `stdio: "inherit"` — critic pentru MCP
- [ ] `npm/ctxlite/scripts/install.js` nu arunca eroare fatala pe platforme nesuportate
- [ ] Fiecare `npm/ctxlite-*/package.json` are `os` si `cpu` fields corecte
- [ ] `make npm-version VERSION=0.2.0` actualizeaza toate package.json-urile sincron
- [ ] `make npm-install-local && cd npm/ctxlite && npm link && ctxlite --version` afiseaza versiunea corecta
- [ ] `.gitkeep` exista in fiecare `npm/ctxlite-*/bin/`
- [ ] `.npmignore` exista in fiecare `npm/ctxlite-*/`
