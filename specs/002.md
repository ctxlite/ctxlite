## Upstream detection

Proxy-ul trebuie sa stie la ce API sa forwardeze, detectat din request path:

```
Request path                          → Upstream
─────────────────────────────────────────────────────────
/v1/messages                          → Anthropic
  (cu header anthropic-version)
/v1/chat/completions                  → OpenAI-compatible
/v1/completions                       → OpenAI-compatible
/anthropic/*                          → Anthropic (alias explicit)
/openai/*                             → OpenAI-compatible (alias explicit)
```

**Upstream URLs configurabile:**

```go
type UpstreamConfig struct {
    AnthropicURL string // default: "https://api.anthropic.com"

}
```

Userul poate suprascrie prin flags sau env vars (pentru alte API-uri OpenAI-compatible — ex: OpenRouter, Groq).

## `proxy.go` — HTTP Server

```go
package proxy

import (
    "context"
    "fmt"
    "net"
    "net/http"
    "time"

    "github.com/ctxlite/ctxlite/go/internal/stats"
)

// Config holds all proxy configuration.
type Config struct {
    Port     int
    Upstream UpstreamConfig
    Store    *stats.Store
    MaxBodyBytes int64 // default: 10MB
}

// Proxy is the HTTP reverse proxy server.
type Proxy struct {
    config Config
    server *http.Server
}

// New creates a new Proxy. Returns error if config is invalid.
func New(config Config) (*Proxy, error) {
    if config.MaxBodyBytes == 0 {
        config.MaxBodyBytes = 10 * 1024 * 1024 // 10MB
    }
    if config.Upstream.AnthropicURL == "" {
        config.Upstream.AnthropicURL = "https://api.anthropic.com"
    }
    if config.Upstream.OpenAIURL == "" {
        config.Upstream.OpenAIURL = "https://api.openai.com"
    }

    p := &Proxy{config: config}

    mux := http.NewServeMux()
    mux.Handle("/", p.applyMiddleware(http.HandlerFunc(p.handle)))

    p.server = &http.Server{
        Addr:         fmt.Sprintf("127.0.0.1:%d", config.Port),
        Handler:      mux,
        ReadTimeout:  30 * time.Second,
        WriteTimeout: 120 * time.Second, // mai mare pentru streaming
        IdleTimeout:  60 * time.Second,
    }

    return p, nil
}

// Start begins listening. Blocks until Stop() is called.
func (p *Proxy) Start() error {
    ln, err := net.Listen("tcp", p.server.Addr)
    if err != nil {
        return fmt.Errorf("proxy: listen on %s: %w", p.server.Addr, err)
    }
    return p.server.Serve(ln)
}

// Stop gracefully shuts down with a 5s timeout.
func (p *Proxy) Stop() {
    ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
    defer cancel()
    _ = p.server.Shutdown(ctx)
}
```

## `upstream.go` — Upstream Detection + Forward

```go
package proxy

import (
    "bytes"
    "fmt"
    "io"
    "net/http"
    "strings"
    "time"
)

// Upstream represents a detected upstream API endpoint.
type Upstream struct {
    Name    string // "anthropic" | "openai"
    BaseURL string
}

// detectUpstream determines which API to forward to based on path and headers.
func (p *Proxy) detectUpstream(r *http.Request) (Upstream, error) {
    path := r.URL.Path

    // Alias explicite
    if strings.HasPrefix(path, "/anthropic/") {
        return Upstream{"anthropic", p.config.Upstream.AnthropicURL}, nil
    }
    if strings.HasPrefix(path, "/openai/") {
        return Upstream{"openai", p.config.Upstream.OpenAIURL}, nil
    }

    // Detectie din path + header
    if path == "/v1/messages" || r.Header.Get("anthropic-version") != "" {
        return Upstream{"anthropic", p.config.Upstream.AnthropicURL}, nil
    }
    if path == "/v1/chat/completions" || path == "/v1/completions" {
        return Upstream{"openai", p.config.Upstream.OpenAIURL}, nil
    }

    return Upstream{}, fmt.Errorf("cannot detect upstream for path: %s", path)
}

// upstreamClient este singletonul HTTP client pentru requests outbound.
// Separat de server pentru a putea seta timeout-uri diferite.
var upstreamClient = &http.Client{
    Timeout: 120 * time.Second,
}

// forward sends the request to the upstream API and returns the raw response.
func (p *Proxy) forward(
    r *http.Request,
    upstream Upstream,
    body []byte,
) (respBody []byte, respHeaders http.Header, statusCode int, err error) {

    // Construieste URL-ul upstream
    targetURL := upstream.BaseURL + r.URL.Path
    if r.URL.RawQuery != "" {
        targetURL += "?" + r.URL.RawQuery
    }

    req, err := http.NewRequestWithContext(r.Context(), r.Method, targetURL, bytes.NewReader(body))
    if err != nil {
        return nil, nil, 0, fmt.Errorf("build upstream request: %w", err)
    }

    // Copiaza headerele originale (inclusiv Authorization / x-api-key)
    // SECURITATE: le forwardam, nu le stocam
    copyHeaders(req.Header, r.Header)

    // Seteaza Host corect
    req.Header.Set("Host", req.URL.Host)

    resp, err := upstreamClient.Do(req)
    if err != nil {
        return nil, nil, 0, fmt.Errorf("upstream call: %w", err)
    }
    defer resp.Body.Close()

    // Verifica daca e streaming
    if isStreamingResponse(resp) {
        // Streaming se gestioneaza separat — vezi stream.go
        // Pentru acum returnam eroare clara ca reminder
        return nil, nil, 0, fmt.Errorf("streaming response detected — implement stream.go (SPEC-02)")
    }

    respBody, err = io.ReadAll(resp.Body)
    if err != nil {
        return nil, nil, 0, fmt.Errorf("read upstream response: %w", err)
    }

    return respBody, resp.Header, resp.StatusCode, nil
}

// isStreamingResponse returns true if the response is an SSE stream.
func isStreamingResponse(resp *http.Response) bool {
    ct := resp.Header.Get("Content-Type")
    return strings.Contains(ct, "text/event-stream")
}

// copyHeaders copies headers from src to dst, excluding hop-by-hop headers.
func copyHeaders(dst, src http.Header) {
    hopByHop := map[string]bool{
        "Connection":          true,
        "Keep-Alive":          true,
        "Proxy-Authenticate":  true,
        "Proxy-Authorization": true,
        "Te":                  true,
        "Trailers":            true,
        "Transfer-Encoding":   true,
        "Upgrade":             true,
    }
    for k, vv := range src {
        if hopByHop[k] {
            continue
        }
        for _, v := range vv {
            dst.Add(k, v)
        }
    }
}
```

## `stream.go` — SSE Streaming Passthrough

Streaming este **obligatoriu** — Cursor si Claude Code folosesc streaming by default. Fara acesta, tool-ul nu functioneaza.

```go
package proxy

import (
    "bufio"
    "fmt"
    "io"
    "log/slog"
    "net/http"
    "time"

    "github.com/ctxlite/ctxlite/go/internal/stats"
)

// handleStream forwards an SSE stream from upstream to the client.
// It passes through all events unchanged and collects stats at the end.
func (p *Proxy) handleStream(
    w http.ResponseWriter,
    r *http.Request,
    upstream Upstream,
    body []byte,
    start time.Time,
) {
    // Verifica ca clientul suporta flushing
    flusher, ok := w.(http.Flusher)
    if !ok {
        http.Error(w, "streaming not supported", http.StatusInternalServerError)
        return
    }

    // Construieste request upstream (identic cu forward())
    targetURL := upstream.BaseURL + r.URL.Path
    req, err := http.NewRequestWithContext(r.Context(), r.Method, targetURL, nil)
    if err != nil {
        http.Error(w, "upstream request error", http.StatusBadGateway)
        return
    }
    req.Body = io.NopCloser(bytesReader(body))
    copyHeaders(req.Header, r.Header)

    resp, err := upstreamClient.Do(req)
    if err != nil {
        http.Error(w, "upstream error", http.StatusBadGateway)
        return
    }
    defer resp.Body.Close()

    // Seteaza headers SSE pe response
    copyHeaders(w.Header(), resp.Header)
    w.Header().Set("Cache-Control", "no-cache")
    w.Header().Set("Connection", "keep-alive")
    w.WriteHeader(resp.StatusCode)
    flusher.Flush()

    // Passthrough linie cu linie
    scanner := bufio.NewScanner(resp.Body)
    for scanner.Scan() {
        line := scanner.Text()
        fmt.Fprintf(w, "%s\n", line)
        flusher.Flush()

        // [HOOK] SPEC-04: acumuleaza token counts din events
        // parseStreamEvent(line, &tokenCounter)
    }

    if err := scanner.Err(); err != nil && err != io.EOF {
        slog.Warn("stream interrupted", "err", err)
    }

    // Log dupa ce stream-ul s-a terminat
    p.logRequest(r.Context(), stats.RequestLog{
        Upstream:  upstream.Name,
        CacheHit:  false,
        Streamed:  true,
        LatencyMs: time.Since(start).Milliseconds(),
    })
}

// bytesReader wraps []byte as io.Reader pentru reuse
func bytesReader(b []byte) io.Reader {
    return &bytesReadCloser{data: b, pos: 0}
}

type bytesReadCloser struct {
    data []byte
    pos  int
}

func (b *bytesReadCloser) Read(p []byte) (int, error) {
    if b.pos >= len(b.data) {
        return 0, io.EOF
    }
    n := copy(p, b.data[b.pos:])
    b.pos += n
    return n, nil
}
```

## `middleware.go`

```go
package proxy

import (
    "log/slog"
    "net/http"
    "runtime/debug"
    "time"
)

// applyMiddleware wraps handler with logging and panic recovery.
func (p *Proxy) applyMiddleware(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        start := time.Now()

        // Panic recovery — nu crapa intregul proxy pe un request rau
        defer func() {
            if rec := recover(); rec != nil {
                slog.Error("panic in handler", "err", rec, "stack", string(debug.Stack()))
                http.Error(w, "internal error", http.StatusInternalServerError)
            }
        }()

        // SECURITATE: nu loga niciodata Authorization sau x-api-key
        slog.Debug("request",
            "method", r.Method,
            "path", r.URL.Path,
            "content_type", r.Header.Get("Content-Type"),
        )

        // Valideaza Content-Type pentru POST
        if r.Method == http.MethodPost {
            ct := r.Header.Get("Content-Type")
            if ct == "" {
                http.Error(w, "Content-Type required", http.StatusBadRequest)
                return
            }
        }

        next.ServeHTTP(w, r)

        slog.Debug("response", "path", r.URL.Path, "latency_ms", time.Since(start).Milliseconds())
    })
}
```

// RequestLog holds data for a single proxied request.
type RequestLog struct {

// LogRequest persists a request log entry.
func (s *Store) LogRequest(ctx context.Context, log RequestLog) {
    // INSERT in requests table
    // Genereaza ID cu time.Now().UnixNano() sau uuid simplu

    _, err := s.db.ExecContext(ctx, `

        (id, ts, model, cache_hit, cache_level, similarity,
         tokens_in, tokens_used, tokens_out, cost_saved, latency_ms)

        log.Similarity, log.TokensIn, log.TokensUsed, log.TokensOut,
        log.CostSaved, log.LatencyMs,
    )
    if err != nil {
        slog.Warn("failed to log request", "err", err)
