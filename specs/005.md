## Context

MCP (Model Context Protocol) este protocolul prin care Cursor, OpenCode si Claude Code comunica cu tool servers externe.  
Transport: **stdio** — serverul citeste JSON-RPC de pe `stdin` si scrie raspunsuri pe `stdout`.  
Logging: **exclusiv pe `stderr`** — `stdout` e rezervat pentru protocol.

Aceasta spec implementeaza un MCP server minimal, **fara dependinte externe** — protocolul e JSON-RPC 2.0 simplu.

---

## Protocol MCP — ce trebuie sa implementam

MCP e JSON-RPC 2.0 peste stdio. Avem nevoie doar de 4 metode:

```
Client → Server: initialize
Server → Client: result (capabilities)

Client → Server: notifications/initialized
Server → Client: (no response needed)

Client → Server: tools/list
Server → Client: result (lista de tools)

Client → Server: tools/call  { name, arguments }
Server → Client: result (content array)
```

Fiecare mesaj e un JSON pe o singura linie (`\n` delimiter).

## `protocol.go` — JSON-RPC 2.0 Types

```go
package mcp

import "encoding/json"

// ── JSON-RPC 2.0 ──────────────────────────────────────────────

// Request reprezinta un mesaj JSON-RPC de la client.
type Request struct {
    JSONRPC string          `json:"jsonrpc"`
    ID      *json.RawMessage `json:"id,omitempty"` // null pentru notifications
    Method  string          `json:"method"`
    Params  json.RawMessage `json:"params,omitempty"`
}

// Response reprezinta un raspuns JSON-RPC catre client.
type Response struct {
    JSONRPC string           `json:"jsonrpc"`
    ID      *json.RawMessage `json:"id,omitempty"`
    Result  interface{}      `json:"result,omitempty"`
    Error   *RPCError        `json:"error,omitempty"`
}

// RPCError reprezinta o eroare JSON-RPC.
type RPCError struct {
    Code    int    `json:"code"`
    Message string `json:"message"`
}

// Coduri de eroare standard JSON-RPC 2.0
const (
    ErrParse          = -32700
    ErrInvalidRequest = -32600
    ErrMethodNotFound = -32601
    ErrInvalidParams  = -32602
    ErrInternal       = -32603
)

// ── MCP specific types ──────────────────────────────────────────

// InitializeParams vine de la client la startup.
type InitializeParams struct {
    ProtocolVersion string      `json:"protocolVersion"`
    Capabilities    interface{} `json:"capabilities"`
    ClientInfo      struct {
        Name    string `json:"name"`
        Version string `json:"version"`
    } `json:"clientInfo"`
}

// InitializeResult e raspunsul la initialize.
type InitializeResult struct {
    ProtocolVersion string       `json:"protocolVersion"`
    Capabilities    Capabilities `json:"capabilities"`
    ServerInfo      ServerInfo   `json:"serverInfo"`
}

// Capabilities descrie ce suporta serverul.
type Capabilities struct {
    Tools *ToolsCapability `json:"tools,omitempty"`
}

// ToolsCapability indica suport pentru tools.
type ToolsCapability struct {
    ListChanged bool `json:"listChanged"`
}

// ServerInfo identifica serverul.
type ServerInfo struct {
    Name    string `json:"name"`
    Version string `json:"version"`
}

// ── Tool types ─────────────────────────────────────────────────

// Tool descrie un MCP tool cu schema JSON.
type Tool struct {
    Name        string      `json:"name"`
    Description string      `json:"description"`
    InputSchema InputSchema `json:"inputSchema"`
}

// InputSchema e schema JSON pentru parametrii tool-ului.
type InputSchema struct {
    Type       string                 `json:"type"`
    Properties map[string]SchemaField `json:"properties"`
    Required   []string               `json:"required,omitempty"`
}

// SchemaField descrie un camp din schema.
type SchemaField struct {
    Type        string      `json:"type"`
    Description string      `json:"description"`
    Items       *SchemaItem `json:"items,omitempty"` // pentru arrays
    Enum        []string    `json:"enum,omitempty"`
}

// SchemaItem descrie elementele unui array.
type SchemaItem struct {
    Type       string                 `json:"type"`
    Properties map[string]SchemaField `json:"properties,omitempty"`
}

// ToolsListResult e raspunsul la tools/list.
type ToolsListResult struct {
    Tools []Tool `json:"tools"`
}

// ToolCallParams vin de la client la tools/call.
type ToolCallParams struct {
    Name      string          `json:"name"`
    Arguments json.RawMessage `json:"arguments"`
}

// ToolCallResult e raspunsul la tools/call.
type ToolCallResult struct {
    Content []ContentBlock `json:"content"`
    IsError bool           `json:"isError,omitempty"`
}

// ContentBlock reprezinta un bloc de continut in raspuns.
type ContentBlock struct {
    Type string `json:"type"` // "text" in v1
    Text string `json:"text"`
}

// newTextResult construieste un ToolCallResult cu un singur text block.
func newTextResult(text string) ToolCallResult {
    return ToolCallResult{
        Content: []ContentBlock{{Type: "text", Text: text}},
    }
}

// newErrorResult construieste un ToolCallResult de eroare.
func newErrorResult(msg string) ToolCallResult {
    return ToolCallResult{
        Content: []ContentBlock{{Type: "text", Text: "Error: " + msg}},
        IsError: true,
    }
}
```

## `handlers.go` — Implementarea Tools

```go
package mcp

import (
    "context"
    "encoding/json"
    "fmt"
    "strings"
    "time"

    "github.com/ctxlite/ctxlite/go/internal/stats"
    "github.com/ctxlite/ctxlite/go/internal/trimmer"
)

// Deps contine dependintele necesare handler-elor.
type Deps struct {
    Trimmer *trimmer.Trimmer
    Store   *stats.Store
}

// ── trim_context ───────────────────────────────────────────────

// TrimContextArgs sunt argumentele pentru tool-ul trim_context.
type TrimContextArgs struct {
    Files []struct {
        Path     string `json:"path"`
        Content  string `json:"content"`
        Language string `json:"language"`
    } `json:"files"`
    Query     string `json:"query"`
    MaxTokens int    `json:"max_tokens"`
}

// handleTrimContext implementeaza tool-ul trim_context.
func handleTrimContext(ctx context.Context, raw json.RawMessage, deps Deps) ToolCallResult {
    var args TrimContextArgs
    if err := json.Unmarshal(raw, &args); err != nil {
        return newErrorResult("invalid arguments: " + err.Error())
    }

    if len(args.Files) == 0 {
        return newErrorResult("files array is empty")
    }
    if args.Query == "" {
        return newErrorResult("query is required")
    }

    // Construieste CodeFile-uri pentru trimmer
    files := make([]trimmer.CodeFile, len(args.Files))
    for i, f := range args.Files {
        files[i] = trimmer.CodeFile{
            Path:     f.Path,
            Content:  f.Content,
            Language: f.Language,
            Tokens:   len(f.Content) / 4, // estimare initiala
        }
    }

    // Configureaza max tokens
    maxTokens := 4096
    if args.MaxTokens > 0 {
        maxTokens = args.MaxTokens
    }

    // Scoreaza fisierele cu BM25
    docs := make([]string, len(files))
    for i, f := range files {
        docs[i] = f.Path + " " + f.Content
    }

    scorer := trimmer.NewBM25(docs, trimmer.DefaultBM25Params)
    scores := scorer.ScoreAll(args.Query)

    // Import graph boost
    graph := trimmer.BuildImportGraph(files)
    scoreMap := make(map[string]float64)
    for _, sd := range scores {
        scoreMap[files[sd.Index].Path] = sd.Score
    }
    importBoosts := trimmer.ImportBoost(graph, scoreMap)

    // Selectie
    selection := trimmer.SelectFiles(files, scores, importBoosts, maxTokens, 0.1)

    // Construieste raspunsul text
    var sb strings.Builder

    sb.WriteString(fmt.Sprintf("## trim_context results\n\n"))
    sb.WriteString(fmt.Sprintf("**Files:** %d/%d selected\n", len(selection.Selected), len(files)))
    sb.WriteString(fmt.Sprintf("**Tokens:** %d → %d (saved %d, -%.0f%%)\n\n",
        selection.TokensUsed+selection.TokensSaved,
        selection.TokensUsed,
        selection.TokensSaved,
        selection.TrimRatio*100,
    ))

    sb.WriteString("### Selected files (by relevance)\n\n")
    for i, f := range selection.Selected {
        // Gaseste scorul pentru acest fisier
        score := 0.0
        for _, sd := range scores {
            if files[sd.Index].Path == f.Path {
                score += sd.Score
            }
        }
        score += importBoosts[f.Path]

        sb.WriteString(fmt.Sprintf("%d. `%s` — score: %.2f | %d tokens\n",
            i+1, f.Path, score, f.Tokens))
    }

    if len(selection.Selected) < len(files) {
        sb.WriteString("\n### Excluded files\n\n")
        selectedPaths := make(map[string]bool)
        for _, f := range selection.Selected {
            selectedPaths[f.Path] = true
        }
        for _, f := range files {
            if !selectedPaths[f.Path] {
                sb.WriteString(fmt.Sprintf("- `%s` (%.0f tokens)\n", f.Path, float64(f.Tokens)))
            }
        }
    }

    return newTextResult(sb.String())
}

// ── get_stats ──────────────────────────────────────────────────

// GetStatsArgs sunt argumentele pentru tool-ul get_stats.
type GetStatsArgs struct {
    Period string `json:"period"` // "session" | "today" | "7d" | "30d" | "all"
}

// handleGetStats implementeaza tool-ul get_stats.
func handleGetStats(ctx context.Context, raw json.RawMessage, deps Deps) ToolCallResult {
    var args GetStatsArgs
    if err := json.Unmarshal(raw, &args); err != nil {
        args.Period = "session" // default
    }
    if args.Period == "" {
        args.Period = "session"
    }

    // Calculeaza intervalul de timp
    var since time.Time
    switch args.Period {
    case "session":
        since = deps.Store.SessionStart()
    case "today":
        now := time.Now()
        since = time.Date(now.Year(), now.Month(), now.Day(), 0, 0, 0, 0, now.Location())
    case "7d":
        since = time.Now().AddDate(0, 0, -7)
    case "30d":
        since = time.Now().AddDate(0, 0, -30)
    case "all":
        since = time.Time{} // zero time = no filter
    default:
        since = deps.Store.SessionStart()
    }

    summary := deps.Store.SummaryFrom(since)

    // Construieste raspunsul formatat
    period := periodLabel(args.Period, since)
    text := formatStatsSummary(summary, period)

    return newTextResult(text)
}

// formatStatsSummary construieste textul formatat pentru statistici.
func formatStatsSummary(s stats.Summary, period string) string {
    hitRate := 0.0
    if s.TotalRequests > 0 {
        hitRate = float64(s.CacheHits) / float64(s.TotalRequests) * 100
    }

    trimRate := 0.0
    if s.TotalRequests-s.CacheHits > 0 {
        trimRate = float64(s.TrimmedRequests) / float64(s.TotalRequests-s.CacheHits) * 100
    }

    return fmt.Sprintf(`## ctxlite stats — %s

**Requests**
- Total: %d
- Cache hits: %d (%.1f%%)
  - L1 (exact): %d
  - L2 (semantic): %d
- Trimmed: %d (%.1f%% of misses)

**Tokens saved**
- From cache: %s
- From trimming: %s
- Total: %s

**Estimated cost saved**
- $%.4f (based on ~$3/M tokens)

**Performance**
- Avg latency (hits): <1ms
- Avg latency (misses): %dms
`,
        period,
        s.TotalRequests,
        s.CacheHits, hitRate,
        s.L1Hits,
        s.L2Hits,
        s.TrimmedRequests, trimRate,
        formatTokens(s.TokensSavedCache),
        formatTokens(s.TokensSavedTrim),
        formatTokens(s.TokensSavedCache+s.TokensSavedTrim),
        s.CostSaved,
        s.AvgLatencyMs,
    )
}

func formatTokens(n int64) string {
    if n >= 1_000_000 {
        return fmt.Sprintf("%.1fM", float64(n)/1_000_000)
    }
    if n >= 1_000 {
        return fmt.Sprintf("%.1fK", float64(n)/1_000)
    }
    return fmt.Sprintf("%d", n)
}

func periodLabel(period string, since time.Time) string {
    switch period {
    case "session":
        return "current session"
    case "today":
        return "today"
    case "7d":
        return "last 7 days"
    case "30d":
        return "last 30 days"
    case "all":
        return "all time"
    }
    return period
}
```

## `server.go` — MCP stdio Server

```go
package mcp

import (
    "bufio"
    "context"
    "encoding/json"
    "fmt"
    "io"
    "log/slog"
    "os"
)

const (
    mcpVersion = "2024-11-05" // versiunea protocolului MCP
    serverName = "ctxlite"
)

// Server este MCP stdio server-ul.
type Server struct {
    deps Deps
    in   io.Reader
    out  io.Writer
}

// New creeaza un MCP Server.
// in/out sunt de obicei os.Stdin / os.Stdout.
// CRITICAL: nu scrie niciodata pe out in afara de handleRequest.
func New(deps Deps, in io.Reader, out io.Writer) *Server {
    return &Server{deps: deps, in: in, out: out}
}

// Serve citeste mesaje JSON-RPC de pe stdin si trimite raspunsuri pe stdout.
// Blocheaza pana cand ctx e cancelled sau stdin e inchis.
func (s *Server) Serve(ctx context.Context) error {
    scanner := bufio.NewScanner(s.in)
    // MCP messages pot fi mari (fisiere in context) — mareste buffer-ul
    buf := make([]byte, 0, 1*1024*1024)  // 1MB initial
    scanner.Buffer(buf, 10*1024*1024)    // max 10MB per linie

    for {
        select {
        case <-ctx.Done():
            return ctx.Err()
        default:
        }

        if !scanner.Scan() {
            if err := scanner.Err(); err != nil {
                return fmt.Errorf("mcp: read error: %w", err)
            }
            return nil // EOF — client a inchis conexiunea
        }

        line := scanner.Bytes()
        if len(line) == 0 {
            continue
        }

        var req Request
        if err := json.Unmarshal(line, &req); err != nil {
            slog.Warn("mcp: invalid JSON", "err", err)
            s.writeError(nil, ErrParse, "parse error")
            continue
        }

        s.handleRequest(ctx, &req)
    }
}

// handleRequest dispatches JSON-RPC request la handler-ul corespunzator.
func (s *Server) handleRequest(ctx context.Context, req *Request) {
    // Notifications (fara ID) nu au nevoie de raspuns
    if req.ID == nil {
        slog.Debug("mcp: notification", "method", req.Method)
        return
    }

    switch req.Method {
    case "initialize":
        s.handleInitialize(req)

    case "tools/list":
        s.writeResult(req.ID, ToolsListResult{Tools: availableTools})

    case "tools/call":
        var params ToolCallParams
        if err := json.Unmarshal(req.Params, &params); err != nil {
            s.writeError(req.ID, ErrInvalidParams, "invalid params")
            return
        }
        result := s.dispatchTool(ctx, params)
        s.writeResult(req.ID, result)

    case "ping":
        s.writeResult(req.ID, map[string]string{})

    default:
        slog.Debug("mcp: unknown method", "method", req.Method)
        s.writeError(req.ID, ErrMethodNotFound, "method not found: "+req.Method)
    }
}

// handleInitialize raspunde la mesajul de initialize.
func (s *Server) handleInitialize(req *Request) {
    var params InitializeParams
    // Ignora erorile de parse la initialize — clientul poate trimite extra fields
    _ = json.Unmarshal(req.Params, &params)

    slog.Debug("mcp: client connected",
        "client", params.ClientInfo.Name,
        "version", params.ClientInfo.Version,
    )

    s.writeResult(req.ID, InitializeResult{
        ProtocolVersion: mcpVersion,
        Capabilities: Capabilities{
            Tools: &ToolsCapability{ListChanged: false},
        },
        ServerInfo: ServerInfo{
            Name:    serverName,
            Version: "0.1.0",
        },
    })
}

// dispatchTool apeleaza handler-ul corespunzator pentru un tool call.
func (s *Server) dispatchTool(ctx context.Context, params ToolCallParams) ToolCallResult {
    slog.Debug("mcp: tool call", "name", params.Name)

    switch params.Name {
    case "trim_context":
        return handleTrimContext(ctx, params.Arguments, s.deps)
    case "get_stats":
        return handleGetStats(ctx, params.Arguments, s.deps)
    default:
        return newErrorResult(fmt.Sprintf("unknown tool: %s", params.Name))
    }
}

// ── I/O helpers ────────────────────────────────────────────────

// writeResult serializes si trimite un raspuns de succes.
// CRITICAL: scrie pe stdout (s.out), nu pe stderr.
func (s *Server) writeResult(id *json.RawMessage, result interface{}) {
    s.write(Response{
        JSONRPC: "2.0",
        ID:      id,
        Result:  result,
    })
}

// writeError serializes si trimite un raspuns de eroare.
func (s *Server) writeError(id *json.RawMessage, code int, msg string) {
    s.write(Response{
        JSONRPC: "2.0",
        ID:      id,
        Error:   &RPCError{Code: code, Message: msg},
    })
}

// write serializes Response si il scrie pe out cu newline.
// Mutex nu e necesar — Serve() este single-threaded pe I/O.
func (s *Server) write(resp Response) {
    data, err := json.Marshal(resp)
    if err != nil {
        slog.Error("mcp: marshal response failed", "err", err)
        return
    }
    // Newline-delimited JSON — un mesaj per linie
    fmt.Fprintf(s.out, "%s\n", data)
}
```

```go
var sessionStart = time.Now()

// SessionStart returneaza momentul de start al sesiunii curente.
func (s *Store) SessionStart() time.Time {
    return sessionStart
}

// Summary extinde structura existenta cu campuri noi.
type Summary struct {
    TotalRequests   int
    CacheHits       int
    L1Hits          int
    L2Hits          int
    TrimmedRequests int     // NOU
    TokensSavedCache int64  // NOU — tokens salvati prin cache
    TokensSavedTrim  int64  // NOU — tokens salvati prin trimming
    CostSaved       float64
    AvgLatencyMs    int64   // NOU
}

// SummaryFrom returneaza statistici agregate de la momentul `since`.
// since == zero time → toate datele (no filter).
func (s *Store) SummaryFrom(since time.Time) Summary {
    var query string
    var args []interface{}

    if since.IsZero() {
        query = `
            SELECT
                COUNT(*)                              as total,
                SUM(CASE WHEN cache_hit THEN 1 END)  as hits,
                SUM(CASE WHEN cache_level='L1' THEN 1 END) as l1,
                SUM(CASE WHEN cache_level='L2' THEN 1 END) as l2,
                SUM(CASE WHEN tokens_in > tokens_used AND NOT cache_hit THEN 1 END) as trimmed,
                SUM(CASE WHEN cache_hit THEN tokens_in END) as saved_cache,
                SUM(CASE WHEN NOT cache_hit THEN tokens_in - tokens_used END) as saved_trim,
                SUM(cost_saved)                       as cost,
                AVG(CASE WHEN NOT cache_hit THEN latency_ms END) as avg_lat
            FROM requests
        `
    } else {
        query = `
            SELECT
                COUNT(*)                              as total,
                SUM(CASE WHEN cache_hit THEN 1 END)  as hits,
                SUM(CASE WHEN cache_level='L1' THEN 1 END) as l1,
                SUM(CASE WHEN cache_level='L2' THEN 1 END) as l2,
                SUM(CASE WHEN tokens_in > tokens_used AND NOT cache_hit THEN 1 END) as trimmed,
                SUM(CASE WHEN cache_hit THEN tokens_in END) as saved_cache,
                SUM(CASE WHEN NOT cache_hit THEN tokens_in - tokens_used END) as saved_trim,
                SUM(cost_saved)                       as cost,
                AVG(CASE WHEN NOT cache_hit THEN latency_ms END) as avg_lat
            FROM requests
            WHERE ts >= ?
        `
        args = append(args, since.Unix())
    }

    var (
        total, hits, l1, l2, trimmed       sql.NullInt64
        savedCache, savedTrim              sql.NullInt64
        cost                              sql.NullFloat64
        avgLat                            sql.NullFloat64
    )

    row := s.db.QueryRow(query, args...)
    _ = row.Scan(&total, &hits, &l1, &l2, &trimmed,
        &savedCache, &savedTrim, &cost, &avgLat)

    return Summary{
        TotalRequests:   int(total.Int64),
        CacheHits:       int(hits.Int64),
        L1Hits:          int(l1.Int64),
        L2Hits:          int(l2.Int64),
        TrimmedRequests: int(trimmed.Int64),
        TokensSavedCache: savedCache.Int64,
        TokensSavedTrim:  savedTrim.Int64,
        CostSaved:       cost.Float64,
        AvgLatencyMs:    int64(avgLat.Float64),
    }
}
```

```go
// Package mcp implements a Model Context Protocol (MCP) server
// over stdio transport.
//
// The server exposes two tools:
//   - trim_context: reduces file context to the most relevant files
//   - get_stats: returns session-level token savings statistics
//
// Communication uses newline-delimited JSON-RPC 2.0.
// All logging is written to stderr; stdout is reserved for the protocol.
package mcp
```

## `mcp_test.go`

### Test helper — simuleaza client MCP

```go
// mcpPipe creeaza un server MCP cu stdin/stdout simulate prin pipes.
func mcpPipe(t *testing.T, deps mcp.Deps) (send func(string), recv func() string) {
    t.Helper()
    clientIn, serverOut := io.Pipe()   // client citeste, server scrie
    serverIn, clientOut := io.Pipe()   // server citeste, client scrie

    srv := mcp.New(deps, serverIn, serverOut)
    ctx, cancel := context.WithCancel(context.Background())
    t.Cleanup(func() {
        cancel()
        clientOut.Close()
        clientIn.Close()
    })
    go srv.Serve(ctx)

    scanner := bufio.NewScanner(clientIn)

    send = func(msg string) {
        fmt.Fprintf(clientOut, "%s\n", msg)
    }
    recv = func() string {
        if !scanner.Scan() { t.Fatal("no response from MCP server") }
        return scanner.Text()
    }
    return send, recv
}
```

### Test 1: initialize

```go
func TestMCP_Initialize(t *testing.T) {
    send, recv := mcpPipe(t, mcp.Deps{})

    send(`{"jsonrpc":"2.0","id":1,"method":"initialize","params":{"protocolVersion":"2024-11-05","clientInfo":{"name":"cursor","version":"1.0"}}}`)

    raw := recv()
    var resp struct {
        JSONRPC string `json:"jsonrpc"`
        ID      int    `json:"id"`
        Result  struct {
            ProtocolVersion string `json:"protocolVersion"`
            ServerInfo      struct {
                Name string `json:"name"`
            } `json:"serverInfo"`
            Capabilities struct {
                Tools interface{} `json:"tools"`
            } `json:"capabilities"`
        } `json:"result"`
    }
    if err := json.Unmarshal([]byte(raw), &resp); err != nil {
        t.Fatalf("unmarshal: %v — raw: %s", err, raw)
    }
    if resp.Result.ServerInfo.Name != "ctxlite" {
        t.Errorf("got server name %q, want ctxlite", resp.Result.ServerInfo.Name)
    }
    if resp.Result.Capabilities.Tools == nil {
        t.Error("tools capability missing")
    }
}
```

### Test 2: tools/list returneaza cele 2 tools

```go
func TestMCP_ToolsList(t *testing.T) {
    send, recv := mcpPipe(t, mcp.Deps{})

    // Initialize first
    send(`{"jsonrpc":"2.0","id":1,"method":"initialize","params":{}}`)
    recv()

    send(`{"jsonrpc":"2.0","id":2,"method":"tools/list","params":{}}`)
    raw := recv()

    var resp struct {
        Result struct {
            Tools []struct {
                Name string `json:"name"`
            } `json:"tools"`
        } `json:"result"`
    }
    if err := json.Unmarshal([]byte(raw), &resp); err != nil {
        t.Fatalf("unmarshal: %v", err)
    }
    if len(resp.Result.Tools) != 2 {
        t.Errorf("expected 2 tools, got %d", len(resp.Result.Tools))
    }

    names := make(map[string]bool)
    for _, tool := range resp.Result.Tools {
        names[tool.Name] = true
    }
    if !names["trim_context"] { t.Error("trim_context tool missing") }
    if !names["get_stats"] { t.Error("get_stats tool missing") }
}
```

### Test 3: trim_context — fisierul relevant e selectat

```go
func TestMCP_TrimContext_SelectsRelevantFile(t *testing.T) {
    tr := trimmer.New(trimmer.Config{MaxContextTokens: 4096})
    send, recv := mcpPipe(t, mcp.Deps{Trimmer: tr})

    send(`{"jsonrpc":"2.0","id":1,"method":"initialize","params":{}}`)
    recv()

    callMsg := `{
        "jsonrpc":"2.0","id":2,
        "method":"tools/call",
        "params":{
            "name":"trim_context",
            "arguments":{
                "query":"implement user login with JWT",
                "files":[
                    {"path":"auth/login.go","content":"func Login(user, pass string) (Token, error) { return jwt.Sign(user) }","language":"go"},
                    {"path":"render/template.go","content":"func RenderHTML(name string) []byte { return templates[name] }","language":"go"},
                    {"path":"db/migrate.go","content":"func RunMigrations(db *sql.DB) error { return migrate.Up(db) }","language":"go"}
                ]
            }
        }
    }`
    send(callMsg)
    raw := recv()

    var resp struct {
        Result struct {
            Content []struct {
                Text string `json:"text"`
            } `json:"content"`
        } `json:"result"`
    }
    if err := json.Unmarshal([]byte(raw), &resp); err != nil {
        t.Fatalf("unmarshal: %v — raw: %s", err, raw)
    }
    if len(resp.Result.Content) == 0 {
        t.Fatal("empty content in response")
    }

    text := resp.Result.Content[0].Text
    if !strings.Contains(text, "auth/login.go") {
        t.Errorf("expected auth/login.go in selected files, got:\n%s", text)
    }
}
```

```go
func TestMCP_GetStats_ReturnsData(t *testing.T) {
    store, _ := stats.New(":memory:")

    send, recv := mcpPipe(t, mcp.Deps{Store: store})

        t.Errorf("unexpected stats format:\n%s", resp.Result.Content[0].Text)

}
```

```go
        t.Fatalf("unmarshal: %v", err)

    if resp.Error == nil {

        t.Errorf("expected code -32601 (MethodNotFound), got %d", resp.Error.Code)
    }
```

### Test 7: notification (fara ID) nu trimite raspuns

```go
func TestMCP_Notification_NoResponse(t *testing.T) {
    send, recv := mcpPipe(t, mcp.Deps{})

    // Trimite notification (fara id)
    send(`{"jsonrpc":"2.0","method":"notifications/initialized","params":{}}`)

    // Trimite imediat un request valid
    send(`{"jsonrpc":"2.0","id":99,"method":"tools/list","params":{}}`)

    raw := recv()

    // Primul (si singurul) raspuns trebuie sa fie pentru tools/list (id=99)
    var resp struct {
        ID int `json:"id"`
    }
    if err := json.Unmarshal([]byte(raw), &resp); err != nil {
        t.Fatalf("unmarshal: %v", err)
    }
    if resp.ID != 99 {
        t.Errorf("expected response id=99, got id=%d — notification should not generate a response", resp.ID)
    }
}
```
